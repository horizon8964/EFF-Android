cmake_minimum_required(VERSION 3.22)
project(EFFAndroidProbe LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
get_filename_component(EFF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${EFF_ROOT}/third_party/dobby" "${CMAKE_CURRENT_BINARY_DIR}/dobby" EXCLUDE_FROM_ALL)
get_target_property(DOBBY_SOURCES dobby_static SOURCES)
list(FILTER DOBBY_SOURCES EXCLUDE REGEX "code-patch-tool-posix\\.cc$")
set_property(TARGET dobby_static PROPERTY SOURCES "${DOBBY_SOURCES}")
target_sources(dobby_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/code_patch.cpp")
add_library(eff_android_probe SHARED runtime_probe.cpp runtime_loader.cpp hot_loader.cpp)
target_compile_definitions(eff_android_probe PRIVATE EFF_ANDROID_VERSION="0.8.2")
target_include_directories(eff_android_probe PRIVATE "${EFF_ROOT}/third_party/eff/src")
target_compile_features(eff_android_probe PRIVATE cxx_std_17)
target_compile_options(eff_android_probe PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(eff_android_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_link_libraries(eff_android_probe PRIVATE dobby_static log dl)

# Offline author-resource compiler: no Unity calls and no injected hooks.
add_executable(eff_compile compiler_main.cpp)
target_compile_features(eff_compile PRIVATE cxx_std_17)
target_compile_options(eff_compile PRIVATE -O2 -fno-autolink -Wno-unknown-pragmas)
target_include_directories(eff_compile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/portable" "${EFF_ROOT}/third_party/eff/src")

add_library(eff_compiler STATIC compiler.cpp)
set_target_properties(eff_compiler PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(eff_compiler PRIVATE cxx_std_17)
target_compile_options(eff_compiler PRIVATE -O2 -fno-autolink -Wno-unknown-pragmas -fvisibility=hidden)
target_include_directories(eff_compiler PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/portable" "${EFF_ROOT}/third_party/eff/src")
target_link_libraries(eff_android_probe PRIVATE eff_compiler)
